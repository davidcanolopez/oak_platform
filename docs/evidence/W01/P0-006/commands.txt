        mermaid_errors.append("Invalid flowchart header")
    sub_count = sum(1 for ln in lines if ln.startswith("subgraph "))
    end_count = sum(1 for ln in lines if ln == "end")
    if sub_count < 1 or sub_count != end_count:
        mermaid_errors.append(f"Unbalanced subgraphs: {sub_count} vs {end_count}")
