Comando reproducible (Python 3, lectura UTF-8):
```powershell
@'
import hashlib, re
from pathlib import Path

req_text = Path("docs/agentic/REQUIREMENTS_BASELINE.md").read_text(encoding="utf-8")

expected_acs = sorted(set(re.findall(r"\bAC-J\d{2}-\d{2}\b", req_text)))
expected_nfrs = sorted(set(re.findall(r"\bNFR-\d{2}\b", req_text)))

found_nfrs = sorted(set(re.findall(r"\bNFR-\d{2}\b", v1_text)))

assert len(expected_nfrs) == 9 and expected_nfrs == found_nfrs

banned = [
    "david cano",
    "trail del robledal",
    "pedales de guadarrama",
    "10k madrid rio",
    "sierra norte trail",
    "oak.run",
    "explorer",
    "pathfinder",
]
v1_lines = v1_text.splitlines()
s53_start = next(i for i, l in enumerate(v1_lines) if l.startswith("### 5.3 "))

outside_53 = "\n".join(v1_lines[:s53_start] + v1_lines[s6_start:]).lower()
inside_53 = "\n".join(v1_lines[s53_start:s6_start]).lower()

assert not any(t in outside_53 for t in banned)
assert all(t in inside_53 for t in banned)

email_re = re.compile(r"(?<![/\w.-])[A-Za-z0-9._%+-]+@(?!\d+\.\d+)[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b")
```
