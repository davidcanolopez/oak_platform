## 2. Controles documentales, no contaminación y SHA-256 de `sites/**` (tras la última edición)
Comando reproducible (Python 3, lectura UTF-8):
```powershell
@'
import hashlib, re, unicodedata
from pathlib import Path

def norm(text: str) -> str:
    return "".join(
        ch for ch in unicodedata.normalize("NFD", text)
        if unicodedata.category(ch) != "Mn"
    ).lower()

req_text = Path("docs/agentic/REQUIREMENTS_BASELINE.md").read_text(encoding="utf-8")
doc_text = Path("docs/product/PERSONAS_AND_DATA_CATALOG.md").read_text(encoding="utf-8")

expected_acs = sorted(set(re.findall(r"\bAC-J\d{2}-\d{2}\b", req_text)))
expected_nfrs = sorted(set(re.findall(r"\bNFR-\d{2}\b", req_text)))
found_acs = sorted(set(re.findall(r"\bAC-J\d{2}-\d{2}\b", doc_text)))
found_nfrs = sorted(set(re.findall(r"\bNFR-\d{2}\b", doc_text)))

assert len(expected_acs) == 44 and expected_acs == found_acs
assert len(expected_nfrs) == 9 and expected_nfrs == found_nfrs

entities = [
    "User", "AthleteProfile", "ConsentRecord", "Organization", "OrganizationMembership",
    "Event", "EventModality", "PriceTier", "EventMedia", "Favourite",
    "Registration", "RegistrationParticipant", "Payment", "PaymentWebhookEvent",
    "CommunicationLog", "AuditLog"
]
assert all(re.search(rf"\b{re.escape(e)}\b", doc_text) for e in entities)

classes = [
    "N1 Pública", "N2 Interna", "N3 Confidencial", "N4 PII", "N5 Financiera/Pago", "N6 Secreto"
]
assert all(c in doc_text for c in classes)

d1_09_count = doc_text.count("blocked_by_D1_09")
hipotesis_count = doc_text.count("HIPOTESIS_DE_PRODUCTO")
assert d1_09_count > 0 and hipotesis_count > 0

schema_text = Path("data/fixtures/schema.ts").read_text(encoding="utf-8")
m = re.search(r"BANNED_DECK_TERMS\s*=\s*\[(.*?)\]\s*as const", schema_text, re.S)
banned = re.findall(r'"([^"]+)"', m.group(1))
assert len(banned) == 8

email_re = re.compile(r"(?<![/\w.-])[A-Za-z0-9._%+-]+@(?!\d+\.\d+)[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b")
allowed = [
    "docs/product/PERSONAS_AND_DATA_CATALOG.md",
    "docs/agentic/packets/P1-002.md",
    "docs/evidence/W02/P1-002/summary.md",
    "docs/evidence/W02/P1-002/commands.txt",
    "docs/evidence/W02/P1-002/approvals.md",
]
for f in allowed:
    raw = Path(f).read_bytes()
    assert b"\r\n" not in raw, f"CRLF in {f}"
    assert not raw.startswith(b"\xef\xbb\xbf"), f"BOM in {f}"
    txt = raw.decode("utf-8")
    assert all(mail.lower().endswith(".invalid") for mail in email_re.findall(txt)), f"Non-.invalid email in {f}"
    norm_txt = norm(txt)
    assert not any(term in norm_txt for term in banned), f"Banned term in {f}"

packet_txt = Path("docs/agentic/packets/P1-002.md").read_text(encoding="utf-8")
assert 'status: "PEER_REVIEW"' in packet_txt
assert "COMPLETAR_" not in packet_txt
summary_txt = Path("docs/evidence/W02/P1-002/summary.md").read_text(encoding="utf-8")
assert "COMPLETAR_" not in summary_txt and "PENDIENTE_MATERIALIZACION" not in summary_txt

manifest_lines = Path("docs/evidence/W01/P0-001/sha256-sites.txt").read_text(encoding="utf-8").splitlines()
matches = 0
for line in manifest_lines:
    if not line.startswith("MATCH"):
        continue
    _, exp_hash, exp_size, rel_path = line.split()
    data = Path(rel_path).read_bytes()
    assert hashlib.sha256(data).hexdigest().lower() == exp_hash.lower()
    assert len(data) == int(exp_size)
    matches += 1
assert matches == 19

print(f"PASS entities=16/16 classes=6/6 AC={len(found_acs)}/44 NFR={len(found_nfrs)}/9 d1_09={d1_09_count} hipotesis={hipotesis_count} non_invalid_emails=0 banned_terms=0 sites_sha256={matches}/19 crlf=0")
'@ | python -
```
